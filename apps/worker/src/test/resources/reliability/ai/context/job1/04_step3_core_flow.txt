- status 를 PROCESSING 으로 변경
- attempt_count 증가

핵심 SQL 조각
WITH cte AS (
  SELECT billing_month_day, user_id
  FROM billing_work
  WHERE billing_month_day = :billingMonthDay
    AND (
      status = 'TARGET'
      OR (status = 'PROCESSING' AND lease_until IS NOT NULL AND lease_until < :now)
    )
  ORDER BY user_id
  LIMIT :limit FOR UPDATE SKIP LOCKED
)
